/* =========================================================
   TIME AGO
   ========================================================= */

function timeAgo(date) {

    const seconds =
        Math.floor((new Date() - date) / 1000);

    let interval =
        Math.floor(seconds / 31536000);

    if (interval >= 1)
        return interval + "y ago";

    interval =
        Math.floor(seconds / 2592000);

    if (interval >= 1)
        return interval + "mo ago";

    interval =
        Math.floor(seconds / 86400);

    if (interval >= 1)
        return interval + "d ago";

    interval =
        Math.floor(seconds / 3600);

    if (interval >= 1)
        return interval + "h ago";

    interval =
        Math.floor(seconds / 60);

    if (interval >= 1)
        return interval + "m ago";

    return "just now";
}


/* =========================================================
   LOAD MEMBERS
   ========================================================= */

async function load() {

    const list =
        document.getElementById("list");

    const headers = {
        "User-Agent":
            "ChessIndianGiantsSidebar/3.0"
    };

    const defaultPawn =
        "https://www.chess.com/bundles/web/images/user-image.007dad08.svg";

    try {

        const response =
            await fetch(
                "https://api.chess.com/pub/club/chess-indian-giants/members",
                { headers }
            );

        const data =
            await response.json();


        /* Combine weekly + monthly */

        const allMembers = [
            ...(data.weekly || []),
            ...(data.monthly || [])
        ];


        /* Newest members first */

        const top3 =
            allMembers
                .sort((a, b) =>
                    b.joined - a.joined
                )
                .slice(0, 3);


        list.innerHTML = "";


        /* Fetch player profiles */

        const memberPromises =
            top3.map(member =>

                fetch(
                    `https://api.chess.com/pub/player/${member.username}`,
                    { headers }
                )

                .then(response =>
                    response.json()
                )

                .then(player => ({

                    username:
                        member.username,

                    avatar:
                        (
                            player.avatar &&
                            player.avatar !== ""
                        )
                        ?
                        player.avatar
                        :
                        defaultPawn,

                    joined:
                        member.joined

                }))

                .catch(error => {

                    console.error(error);

                    return {

                        username:
                            member.username,

                        avatar:
                            defaultPawn,

                        joined:
                            member.joined

                    };

                })

            );


        const members =
            await Promise.all(memberPromises);


        /* Render */

        members.forEach(
            (member, index) => {

                const joinedDate =
                    new Date(
                        member.joined * 1000
                    );


                const memberHTML = `

                    <a
                        href="https://www.chess.com/member/${member.username}"
                        target="_blank"
                    >

                        <div class="member-box">

                            <div class="rank-number">
                                ${index + 1}
                            </div>


                            <div class="avatar-wrapper">

                                <img
                                    src="${member.avatar}"
                                    class="avatar"
                                    onerror="this.src='${defaultPawn}'"
                                >

                                <div class="new-badge">
                                    NEW
                                </div>

                            </div>


                            <div class="info-container">

                                <span class="name">
                                    ${member.username}
                                </span>

                                <span class="time-ago">
                                    ${timeAgo(joinedDate)}
                                </span>

                            </div>

                        </div>

                    </a>

                `;


                list.innerHTML += memberHTML;

            }
        );

    }

    catch (error) {

        console.error(error);

        list.innerHTML =
            "<div class='loading-text'>Unable to load members</div>";

    }

}


/* START */

load();
